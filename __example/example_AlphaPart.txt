rm(list=ls())
library(AlphaPart)
myebvs=read.table("~/Downloads/ped_ebvs_sex.txt", header=T)
# number of generation 
year=c(rep(1,1100))
for (i in 2:11){ year=c(year,rep(i,2000)) }
myebvs$year=year

part <- AlphaPart(myebvs,
                  colPath = "sex",
                  colBV = "EBV")
sumPartByYear <- summary(part, by = "year")
plot(sumPartByYear,
     col=c("red","blue"), main="EBVs OK")

# now sum a constant (20) to EBVs
myebvs$EBV2=myebvs$EBV + 20
part <- AlphaPart(myebvs,
                  colPath = "sex",
                  colBV = "EBV2")
sumPartByYear <- summary(part, by = "year")
plot(sumPartByYear,
     col=c("red","blue"),ylab="EBV-20")

tmp <- myebvs[myebvs$year == 1,]
hist(tmp$EBV2)
